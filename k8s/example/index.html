    apiVersion: v1
kind: Secret
metadata:
  name: mongodb-secret # this name shold match with secret names in deployment
type: Opaque #default for arbitrary key-value pairs
data:
  mongo-root-username: cm9vdA==        # base64 for "root" echo 
  mongo-root-password: ZXhhbXBsZQ==    # base64 for "example"

    apiVersion: apps/v1
kind: Deployment
metadata:
  name: mongodb-deployment
  labels:
    app: mongodb
spec:
  replicas: 1
  selector:
    matchLabels:
      app: mongodb
  template:
    metadata:
      labels:
        app: mongodb
    spec:
      containers: # pod blueprint 
        - name: mongodb
          image: mongo:6.0
          ports:
            - containerPort: 27017 #this port has to match with service target port
          env:
            - name: MONGO_INITDB_ROOT_USERNAME
              # value: root // use this incase direct value instead of secret
              valueFrom:
                secretKeyRef:
                  name: mongodb-secret
                  key: mongo-root-username
            - name: MONGO_INITDB_ROOT_PASSWORD
              # value: example
              valueFrom:
                secretKeyRef:
                  name: mongodb-secret
                  key: mongo-root-password
---
# creating a service
apiVersion: v1
kind: Service
metadata:
  name: mongodb-service
spec:
  selector:
    app: mongodb # to Connect to pod through label
  ports:
    - protocol: TCP
      port: 27017 #service port, this can be different
      targetPort: 27017 #container port of deployment

    apiVersion: v1
kind: ConfigMap
metadata:
  name: mongodb-configmap
data:
  database_url: mongodb-service # name of service

    apiVersion: apps/v1
kind: Deployment
metadata:
  name: mongo-express
  labels:
    app: mongo-express
spec:
  replicas: 1
  selector:
    matchLabels:
      app: mongo-express
  template:
    metadata:
      labels:
        app: mongo-express
    spec:
      containers:
      - name: mongo-express
        image: mongo-express
        ports:
        - containerPort: 8081
        env:
        - name: ME_CONFIG_MONGODB_ADMINUSERNAME
          valueFrom:
            secretKeyRef:
              name: mongodb-secret
              key: mongo-root-username
        - name: ME_CONFIG_MONGODB_ADMINPASSWORD
          valueFrom: 
            secretKeyRef:
              name: mongodb-secret
              key: mongo-root-password
        - name: ME_CONFIG_MONGODB_SERVER
          valueFrom: 
            configMapKeyRef:
              name: mongodb-configmap
              key: database_url
---
apiVersion: v1
kind: Service
metadata:
  name: mongo-express-service
spec:
  selector:
    app: mongo-express
  type: LoadBalancer   # makeing it as external service, assign external IP address to service.
  ports:
    - protocol: TCP
      port: 8081
      targetPort: 8081 # continaer port should be same
      nodePort: 30000 # port for external ip address, port that should be put into browser
